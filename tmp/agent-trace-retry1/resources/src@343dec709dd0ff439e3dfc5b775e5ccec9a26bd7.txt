async function dragRushTask(page: Page, isSaved: () => boolean | Promise<boolean>, pollTimeout = 5000) {
  await expect(page.locator('.gantt-e2e-markers')).toHaveCount(0)

  const rushTask = page.locator('.gantt_task_line[task_id="dhtmlx-item-rush"]').first()
  await expect(rushTask).toBeVisible()
  await rushTask.scrollIntoViewIfNeeded()
  await expect(rushTask).toBeInViewport()

  const dragHandle = rushTask.locator('.gantt_task_content').first()
  await expect(dragHandle).toBeVisible()

  for (let attempt = 0; attempt < 2; attempt += 1) {
    const beforeBox = await waitForStableBox(rushTask)
    const handleBox = await waitForStableBox(dragHandle)
    const startX = handleBox.x + Math.max(12, handleBox.width / 2)
    const startY = handleBox.y + handleBox.height / 2

    await page.mouse.move(startX, startY)
    await page.mouse.down()
    await page.mouse.move(startX + 180, startY, { steps: 16 })
    await page.mouse.up()
    await expect(rushTask).toBeVisible()

    const saved = await expect.poll(() => isSaved(), { timeout: pollTimeout })
      .toBe(true)
      .then(() => true)
      .catch(() => false)
    if (saved) return

    const afterBox = await rushTask.boundingBox()
    const taskMoved = !!afterBox && Math.abs(afterBox.x - beforeBox.x) > 20
    if (!taskMoved) {
      continue
    }
  }

  await expect.poll(() => isSaved(), { timeout: Math.max(10000, pollTimeout) }).toBe(true)
}

function getRescheduleWindow(body: unknown) {
  const payload = body as { plannedStartTime?: unknown; plannedEndTime?: unknown }
  expect(typeof payload.plannedStartTime).toBe('string')
  expect(typeof payload.plannedEndTime).toBe('string')

  const plannedStartTime = new Date(payload.plannedStartTime as string)
  const plannedEndTime = new Date(payload.plannedEndTime as string)
  expect(Number.isNaN(plannedStartTime.getTime())).toBe(false)
  expect(Number.isNaN(plannedEndTime.getTime())).toBe(false)

  return {
    plannedStartTime,
    plannedEndTime,
    durationHours: (plannedEndTime.getTime() - plannedStartTime.getTime()) / (1000 * 60 * 60),
  }
}

async function clickUntilSuggestionAppears(button: Locator, suggestion: Locator) {
  for (let attempt = 0; attempt < 2; attempt += 1) {
    await button.click()
    const appeared = await suggestion
      .filter({ hasText: '急单插入重排建议' })
      .first()
      .waitFor({ state: 'visible', timeout: 2500 })
      .then(() => true)
      .catch(() => false)
    if (appeared) return
  }
  await expect(suggestion).toContainText('急单插入重排建议')
}

async function waitForStableBox(locator: Locator) {
  let stableBox = await locator.boundingBox()
  await expect.poll(async () => {
    const first = await locator.boundingBox()
    if (!first) return false
    await new Promise(resolve => setTimeout(resolve, 100))
    const second = await locator.boundingBox()
    if (!second) return false
    const isStable =
      Math.abs(first.x - second.x) < 1 &&
      Math.abs(first.y - second.y) < 1 &&
      Math.abs(first.width - second.width) < 1 &&
      Math.abs(first.height - second.height) < 1
    if (isStable) stableBox = second
    return isStable
  }, { timeout: 10000 }).toBe(true)
  expect(stableBox).not.toBeNull()
  return stableBox!
}
